    %% Sources -> Middleware
    POS --> ETL
    OMS --> ETL
    CAT --> ETL
    LOG --> ETL
    GA4 --> ETL
    ERP --> ETL

    %% Middleware -> Warehouse
    ETL --> JOIN

    %% Warehouse -> Human Review
    JOIN --> REVIEW

    %% Human Review -> Activation
    REVIEW --> PERS
    REVIEW --> RETGT
    REVIEW --> GMC

    %% Warehouse -> Activation (direct real-time path)
    JOIN -.->|real-time stock / price| PDP

    %% Within Activation
    ADS <-->|feed mapping| GMC
    GMC --> PDP
    RETGT --> PDP
    PERS --> PDP

    %% Activation -> Monitoring
    PDP --> DASH
    GMC --> DASH
    RETGT --> DASH
    PERS --> DASH

    %% Failure handling (reverse / fallback paths)
    JOIN -.->|stale or invalid data| FALLBACK
    DASH -.->|anomaly flagged| REVIEW
